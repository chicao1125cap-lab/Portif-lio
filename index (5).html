public class Veiculo {

    private String motor;

    private String cor;

    public Veiculo(String placa, int ano, String motor,
                   String chassi, String cor, String montadora) {
        this.placa = placa;
        this.ano = ano;
        this.motor = motor;
        this.chassi = chassi;
        this.cor = cor;
        this.montadora = montadora;
    }

    public String getPlaca() {
        return placa;
    }

    public void setPlaca(String placa) {
        this.placa = placa;
    }

    // ...os demais atributos seguem o mesmo padrão de getter/setter
}
GerenciadorVeiculos.java Streams & referências de método 
private final List<Veiculo> veiculos = new ArrayList<>();

public long quantidadeComMaisDeAnos(int anos) {
    return veiculos.stream()
            .filter(v -> v.getAnosDeCirculacao() > anos)
            .count();
}

public double percentualMotor10AnoSuperiorA2000CorBranca() {
    if (veiculos.isEmpty()) {
        return 0.0;
    }
    long quantidade = veiculos.stream()
            .filter(Veiculo::isMotor1_0)
            .filter(v -> v.getAno() > 2000)
            .filter(Veiculo::isCorBranca)
            .count();
    return (quantidade * 100.0) / veiculos.size();
}

public int getAnosDeCirculacao() {
    return Year.now().getValue() - ano;
}

public boolean isMotor1_0() {
    return motor != null && motor.trim().equals("1.0");
}

public boolean isCorBranca() {
    return cor != null && cor.trim().equalsIgnoreCase("branca");
}
